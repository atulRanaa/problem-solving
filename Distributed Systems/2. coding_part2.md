
## 7. Rate Limiter (Token Bucket \& Sliding Window)

### Theory

Rate limiters control request throughput to prevent system overload. **Token Bucket** refills tokens at fixed rate; requests consume tokens. **Sliding Window** counts requests in rolling time window.

**Key Trade-offs**:

- Token Bucket: Allows bursts, simpler implementation
- Sliding Window: More accurate, prevents burst traffic

```cpp
#include <chrono>
#include <mutex>
#include <queue>
#include <atomic>

// Token Bucket Rate Limiter
class TokenBucketRateLimiter {
private:
    using Clock = std::chrono::steady_clock;
    using TimePoint = Clock::time_point;
    
    const size_t capacity;        // Max tokens in bucket
    const double refill_rate;     // Tokens per second
    
    std::atomic<double> tokens;
    std::atomic<TimePoint> last_refill;
    std::mutex mtx;
    
    void refill() {
        TimePoint now = Clock::now();
        TimePoint last = last_refill.load();
        
        auto elapsed = std::chrono::duration_cast<std::chrono::milliseconds>(
            now - last
        ).count();
        
        // Calculate tokens to add
        double tokens_to_add = (elapsed / 1000.0) * refill_rate;
        
        if (tokens_to_add > 0) {
            double current = tokens.load();
            tokens.store(std::min(capacity * 1.0, current + tokens_to_add));
            last_refill.store(now);
        }
    }
    
public:
    TokenBucketRateLimiter(size_t cap, double rate_per_sec)
        : capacity(cap)
        , refill_rate(rate_per_sec)
        , tokens(cap)
        , last_refill(Clock::now()) {}
    
    bool allowRequest() {
        std::lock_guard<std::mutex> lock(mtx);
        
        refill();
        
        double current = tokens.load();
        if (current >= 1.0) {
            tokens.store(current - 1.0);
            return true;
        }
        
        return false;
    }
    
    // Try to consume N tokens atomically
    bool allowRequest(size_t n) {
        std::lock_guard<std::mutex> lock(mtx);
        
        refill();
        
        double current = tokens.load();
        if (current >= n) {
            tokens.store(current - n);
            return true;
        }
        
        return false;
    }
    
    size_t availableTokens() {
        std::lock_guard<std::mutex> lock(mtx);
        refill();
        return static_cast<size_t>(tokens.load());
    }
};

// Sliding Window Rate Limiter
class SlidingWindowRateLimiter {
private:
    using Clock = std::chrono::steady_clock;
    using TimePoint = Clock::time_point;
    
    const size_t max_requests;
    const std::chrono::milliseconds window_size;
    
    // Store timestamps of requests
    std::queue<TimePoint> request_times;
    std::mutex mtx;
    
    void removeExpiredRequests() {
        TimePoint now = Clock::now();
        TimePoint threshold = now - window_size;
        
        while (!request_times.empty() && 
               request_times.front() < threshold) {
            request_times.pop();
        }
    }
    
public:
    SlidingWindowRateLimiter(size_t max_req, std::chrono::seconds window)
        : max_requests(max_req)
        , window_size(std::chrono::duration_cast<std::chrono::milliseconds>(window)) {}
    
    bool allowRequest() {
        std::lock_guard<std::mutex> lock(mtx);
        
        removeExpiredRequests();
        
        if (request_times.size() < max_requests) {
            request_times.push(Clock::now());
            return true;
        }
        
        return false;
    }
    
    size_t currentRequests() {
        std::lock_guard<std::mutex> lock(mtx);
        removeExpiredRequests();
        return request_times.size();
    }
};

// Sliding Window Counter (Memory Optimized)
class SlidingWindowCounterRateLimiter {
private:
    using Clock = std::chrono::steady_clock;
    using TimePoint = Clock::time_point;
    
    const size_t max_requests;
    const std::chrono::seconds window_size;
    
    size_t prev_window_count = 0;
    size_t curr_window_count = 0;
    TimePoint curr_window_start;
    
    std::mutex mtx;
    
public:
    SlidingWindowCounterRateLimiter(size_t max_req, std::chrono::seconds window)
        : max_requests(max_req)
        , window_size(window)
        , curr_window_start(Clock::now()) {}
    
    bool allowRequest() {
        std::lock_guard<std::mutex> lock(mtx);
        
        TimePoint now = Clock::now();
        auto elapsed = std::chrono::duration_cast<std::chrono::seconds>(
            now - curr_window_start
        );
        
        // Move to next window if current expired
        if (elapsed >= window_size) {
            prev_window_count = curr_window_count;
            curr_window_count = 0;
            curr_window_start = now;
            elapsed = std::chrono::seconds(0);
        }
        
        // Calculate weighted count for sliding window
        double prev_weight = 1.0 - (elapsed.count() / (double)window_size.count());
        double estimated_count = prev_window_count * prev_weight + curr_window_count;
        
        if (estimated_count < max_requests) {
            ++curr_window_count;
            return true;
        }
        
        return false;
    }
};
```


## 8. Consistent Hashing

### Theory

Consistent hashing minimizes key remapping when nodes join/leave. Hash both keys and nodes onto a ring [0, 2^32). Key maps to first node clockwise. Virtual nodes improve load distribution.

```cpp
#include <map>
#include <string>
#include <vector>
#include <functional>
#include <algorithm>

class ConsistentHashRing {
private:
    // Ring: hash -> node_id
    std::map<size_t, std::string> ring;
    
    // Number of virtual nodes per physical node
    const int virtual_nodes;
    
    // Hash function
    std::hash<std::string> hasher;
    
    size_t hash(const std::string& key) const {
        return hasher(key);
    }
    
public:
    explicit ConsistentHashRing(int vnodes = 150) 
        : virtual_nodes(vnodes) {}
    
    // Add physical node to ring (creates virtual nodes)
    void addNode(const std::string& node_id) {
        // Create virtual nodes for better distribution
        for (int i = 0; i < virtual_nodes; ++i) {
            std::string virtual_key = node_id + "#" + std::to_string(i);
            size_t hash_value = hash(virtual_key);
            ring[hash_value] = node_id;
        }
    }
    
    // Remove physical node from ring
    void removeNode(const std::string& node_id) {
        for (int i = 0; i < virtual_nodes; ++i) {
            std::string virtual_key = node_id + "#" + std::to_string(i);
            size_t hash_value = hash(virtual_key);
            ring.erase(hash_value);
        }
    }
    
    // Get node responsible for key
    std::string getNode(const std::string& key) const {
        if (ring.empty()) {
            return "";
        }
        
        size_t hash_value = hash(key);
        
        // Find first node clockwise from hash position
        auto it = ring.lower_bound(hash_value);
        
        // Wrap around if at end
        if (it == ring.end()) {
            it = ring.begin();
        }
        
        return it->second;
    }
    
    // Get N replica nodes for key
    std::vector<std::string> getNodes(const std::string& key, size_t n) const {
        std::vector<std::string> result;
        if (ring.empty() || n == 0) {
            return result;
        }
        
        size_t hash_value = hash(key);
        auto it = ring.lower_bound(hash_value);
        
        // Collect unique physical nodes
        std::vector<std::string> seen;
        size_t count = 0;
        
        while (count < n) {
            if (it == ring.end()) {
                it = ring.begin();
            }
            
            const std::string& node = it->second;
            
            // Only add if not already in result
            if (std::find(seen.begin(), seen.end(), node) == seen.end()) {
                seen.push_back(node);
                result.push_back(node);
                ++count;
            }
            
            ++it;
            
            // Prevent infinite loop if fewer nodes than requested
            if (it == ring.end() && count < n && result.size() == countPhysicalNodes()) {
                break;
            }
        }
        
        return result;
    }
    
    size_t countPhysicalNodes() const {
        std::vector<std::string> unique_nodes;
        for (const auto& [hash, node] : ring) {
            if (std::find(unique_nodes.begin(), unique_nodes.end(), node) 
                == unique_nodes.end()) {
                unique_nodes.push_back(node);
            }
        }
        return unique_nodes.size();
    }
    
    size_t countVirtualNodes() const {
        return ring.size();
    }
};
```


## 9. Bloom Filter

### Theory

Probabilistic data structure for membership testing. Uses k hash functions to set k bits. Can have false positives, never false negatives. Space-efficient for large sets.

**Formula**: m = -(n × ln(p)) / (ln(2)²) where m=bits, n=elements, p=false positive rate

```cpp
#include <vector>
#include <functional>
#include <cmath>
#include <string>

class BloomFilter {
private:
    std::vector<bool> bit_array;
    size_t num_hashes;
    size_t bit_count;
    
    // Multiple hash functions using seed variation
    size_t hash(const std::string& item, size_t seed) const {
        std::hash<std::string> hasher;
        size_t h = hasher(item);
        
        // Mix with seed using multiplicative hashing
        h ^= seed + 0x9e3779b9 + (h << 6) + (h >> 2);
        
        return h % bit_count;
    }
    
public:
    // Constructor calculates optimal parameters
    BloomFilter(size_t expected_elements, double false_positive_rate) {
        // Calculate optimal bit array size
        bit_count = static_cast<size_t>(
            -1.0 * expected_elements * std::log(false_positive_rate) 
            / (std::log(2) * std::log(2))
        );
        
        // Calculate optimal number of hash functions
        num_hashes = static_cast<size_t>(
            (bit_count / (double)expected_elements) * std::log(2)
        );
        
        // Ensure at least 1 hash function
        num_hashes = std::max(num_hashes, size_t(1));
        
        bit_array.resize(bit_count, false);
    }
    
    // Direct constructor with specified parameters
    BloomFilter(size_t bits, size_t hashes) 
        : bit_count(bits), num_hashes(hashes) {
        bit_array.resize(bit_count, false);
    }
    
    // Add element to filter
    void add(const std::string& item) {
        for (size_t i = 0; i < num_hashes; ++i) {
            size_t index = hash(item, i);
            bit_array[index] = true;
        }
    }
    
    // Check if element possibly in set
    bool possiblyContains(const std::string& item) const {
        for (size_t i = 0; i < num_hashes; ++i) {
            size_t index = hash(item, i);
            if (!bit_array[index]) {
                return false;  // Definitely not in set
            }
        }
        return true;  // Possibly in set
    }
    
    // Clear all entries
    void clear() {
        std::fill(bit_array.begin(), bit_array.end(), false);
    }
    
    // Get current false positive probability (approximate)
    double currentFalsePositiveRate(size_t items_inserted) const {
        // p = (1 - e^(-k*n/m))^k
        double exponent = -1.0 * num_hashes * items_inserted / bit_count;
        return std::pow(1.0 - std::exp(exponent), num_hashes);
    }
    
    size_t size() const { return bit_count; }
    size_t hashFunctions() const { return num_hashes; }
};

// Counting Bloom Filter (supports deletions)
class CountingBloomFilter {
private:
    std::vector<uint32_t> counters;
    size_t num_hashes;
    size_t counter_count;
    
    std::hash<std::string> hasher;
    
    size_t hash(const std::string& item, size_t seed) const {
        size_t h = hasher(item);
        h ^= seed + 0x9e3779b9 + (h << 6) + (h >> 2);
        return h % counter_count;
    }
    
public:
    CountingBloomFilter(size_t counters, size_t hashes)
        : counter_count(counters), num_hashes(hashes) {
        counters.resize(counter_count, 0);
    }
    
    void add(const std::string& item) {
        for (size_t i = 0; i < num_hashes; ++i) {
            size_t index = hash(item, i);
            if (counters[index] < UINT32_MAX) {
                ++counters[index];
            }
        }
    }
    
    void remove(const std::string& item) {
        for (size_t i = 0; i < num_hashes; ++i) {
            size_t index = hash(item, i);
            if (counters[index] > 0) {
                --counters[index];
            }
        }
    }
    
    bool possiblyContains(const std::string& item) const {
        for (size_t i = 0; i < num_hashes; ++i) {
            size_t index = hash(item, i);
            if (counters[index] == 0) {
                return false;
            }
        }
        return true;
    }
};
```


## 10. Skip List

### Theory

Probabilistic balanced data structure with O(log n) search/insert/delete. Multi-level linked lists where higher levels skip intermediate nodes. Simpler than balanced trees.

**Level Selection**: Coin flip with p=0.5 determines node height.

```cpp
#include <vector>
#include <random>
#include <limits>

template<typename K, typename V>
class SkipList {
private:
    static constexpr int MAX_LEVEL = 16;
    static constexpr double P = 0.5;
    
    struct Node {
        K key;
        V value;
        std::vector<Node*> forward;
        
        Node(K k, V v, int level) 
            : key(k), value(v), forward(level + 1, nullptr) {}
    };
    
    Node* header;
    int current_level;
    std::mt19937 rng;
    std::uniform_real_distribution<double> dist;
    
    // Random level generation using coin flip
    int randomLevel() {
        int level = 0;
        while (dist(rng) < P && level < MAX_LEVEL) {
            ++level;
        }
        return level;
    }
    
public:
    SkipList() : current_level(0), dist(0.0, 1.0) {
        // Header with minimum key
        header = new Node(K{}, V{}, MAX_LEVEL);
        rng.seed(std::random_device{}());
    }
    
    ~SkipList() {
        Node* current = header;
        while (current != nullptr) {
            Node* next = current->forward[^0];
            delete current;
            current = next;
        }
    }
    
    // Search for key
    bool search(const K& key, V& value) const {
        Node* current = header;
        
        // Traverse from top level down
        for (int i = current_level; i >= 0; --i) {
            while (current->forward[i] != nullptr && 
                   current->forward[i]->key < key) {
                current = current->forward[i];
            }
        }
        
        // Move to next node at level 0
        current = current->forward[^0];
        
        if (current != nullptr && current->key == key) {
            value = current->value;
            return true;
        }
        
        return false;
    }
    
    // Insert key-value pair
    void insert(const K& key, const V& value) {
        std::vector<Node*> update(MAX_LEVEL + 1, nullptr);
        Node* current = header;
        
        // Find insertion position and track update path
        for (int i = current_level; i >= 0; --i) {
            while (current->forward[i] != nullptr && 
                   current->forward[i]->key < key) {
                current = current->forward[i];
            }
            update[i] = current;
        }
        
        current = current->forward[^0];
        
        // Update existing node
        if (current != nullptr && current->key == key) {
            current->value = value;
            return;
        }
        
        // Insert new node
        int new_level = randomLevel();
        
        if (new_level > current_level) {
            for (int i = current_level + 1; i <= new_level; ++i) {
                update[i] = header;
            }
            current_level = new_level;
        }
        
        Node* new_node = new Node(key, value, new_level);
        
        // Update forward pointers
        for (int i = 0; i <= new_level; ++i) {
            new_node->forward[i] = update[i]->forward[i];
            update[i]->forward[i] = new_node;
        }
    }
    
    // Delete key
    bool remove(const K& key) {
        std::vector<Node*> update(MAX_LEVEL + 1, nullptr);
        Node* current = header;
        
        // Find node to delete
        for (int i = current_level; i >= 0; --i) {
            while (current->forward[i] != nullptr && 
                   current->forward[i]->key < key) {
                current = current->forward[i];
            }
            update[i] = current;
        }
        
        current = current->forward[^0];
        
        if (current == nullptr || current->key != key) {
            return false;  // Not found
        }
        
        // Update forward pointers to bypass deleted node
        for (int i = 0; i <= current_level; ++i) {
            if (update[i]->forward[i] != current) {
                break;
            }
            update[i]->forward[i] = current->forward[i];
        }
        
        delete current;
        
        // Update current level if necessary
        while (current_level > 0 && 
               header->forward[current_level] == nullptr) {
            --current_level;
        }
        
        return true;
    }
};
```


## 11. Top K Frequent Elements

### Theory

Find K most frequent elements in array. Use hashmap for frequency counting, min-heap of size K to track top K elements. Heap maintains smallest of top K at root.[^12][^13]

**Time**: O(N + N log K) = O(N log K)

```cpp
#include <unordered_map>
#include <queue>
#include <vector>

class TopKFrequent {
public:
    // Using Min-Heap approach
    static std::vector<int> topKFrequent(const std::vector<int>& nums, int k) {
        // Step 1: Count frequencies O(N)
        std::unordered_map<int, int> freq_map;
        for (int num : nums) {
            ++freq_map[num];
        }
        
        // Step 2: Min-heap of size k O(N log k)
        // Heap stores pairs: {frequency, element}
        auto cmp = [](const std::pair<int, int>& a, 
                     const std::pair<int, int>& b) {
            return a.first > b.first;  // Min-heap by frequency
        };
        
        std::priority_queue<
            std::pair<int, int>,
            std::vector<std::pair<int, int>>,
            decltype(cmp)
        > min_heap(cmp);
        
        for (const auto& [num, freq] : freq_map) {
            min_heap.push({freq, num});
            
            // Keep only top k elements
            if (min_heap.size() > k) {
                min_heap.pop();
            }
        }
        
        // Step 3: Extract results
        std::vector<int> result;
        while (!min_heap.empty()) {
            result.push_back(min_heap.top().second);
            min_heap.pop();
        }
        
        return result;
    }
    
    // Bucket sort approach O(N) - better when k is close to unique count
    static std::vector<int> topKFrequentBucketSort(const std::vector<int>& nums, int k) {
        // Count frequencies
        std::unordered_map<int, int> freq_map;
        for (int num : nums) {
            ++freq_map[num];
        }
        
        // Bucket sort: index = frequency, value = list of numbers
        std::vector<std::vector<int>> buckets(nums.size() + 1);
        
        for (const auto& [num, freq] : freq_map) {
            buckets[freq].push_back(num);
        }
        
        // Collect top k from highest frequency buckets
        std::vector<int> result;
        for (int freq = buckets.size() - 1; freq >= 0 && result.size() < k; --freq) {
            for (int num : buckets[freq]) {
                result.push_back(num);
                if (result.size() == k) {
                    return result;
                }
            }
        }
        
        return result;
    }
};
```


## 12. Median from Data Stream

### Theory

Maintain running median using two heaps: max-heap for lower half, min-heap for upper half. Median is at heap roots. Balance heaps to keep sizes within 1.

**Time**: O(log n) insert, O(1) getMedian

```cpp
#include <queue>
#include <vector>

class MedianFinder {
private:
    // Max-heap for lower half (stores negated values)
    std::priority_queue<int> max_heap;
    
    // Min-heap for upper half
    std::priority_queue<int, std::vector<int>, std::greater<int>> min_heap;
    
public:
    MedianFinder() {}
    
    // Add number to data structure
    void addNum(int num) {
        // Always add to max-heap first
        max_heap.push(num);
        
        // Balance: move largest from max-heap to min-heap
        min_heap.push(max_heap.top());
        max_heap.pop();
        
        // Rebalance if min-heap larger
        if (min_heap.size() > max_heap.size()) {
            max_heap.push(min_heap.top());
            min_heap.pop();
        }
    }
    
    // Get current median
    double findMedian() const {
        if (max_heap.size() > min_heap.size()) {
            return max_heap.top();
        }
        
        return (max_heap.top() + min_heap.top()) / 2.0;
    }
    
    size_t size() const {
        return max_heap.size() + min_heap.size();
    }
};

// Generic version for any numeric type
template<typename T>
class GenericMedianFinder {
private:
    std::priority_queue<T> max_heap;
    std::priority_queue<T, std::vector<T>, std::greater<T>> min_heap;
    
public:
    void addNum(T num) {
        if (max_heap.empty() || num <= max_heap.top()) {
            max_heap.push(num);
        } else {
            min_heap.push(num);
        }
        
        // Balance heaps
        if (max_heap.size() > min_heap.size() + 1) {
            min_heap.push(max_heap.top());
            max_heap.pop();
        } else if (min_heap.size() > max_heap.size()) {
            max_heap.push(min_heap.top());
            min_heap.pop();
        }
    }
    
    double findMedian() const {
        if (max_heap.size() == min_heap.size()) {
            return (max_heap.top() + min_heap.top()) / 2.0;
        }
        return max_heap.top();
    }
};
```
