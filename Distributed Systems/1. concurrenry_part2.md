## 7. Thread Pool Implementation

### Theory

Thread pool maintains a fixed set of worker threads that consume tasks from a queue, avoiding expensive thread creation/destruction overhead. Critical for systems like Kafka that process high-throughput workloads.[^1][^2]

**Key Considerations**:

- **Work Stealing**: Advanced pools use work-stealing queues for load balancing
- **Thread Affinity**: Pin threads to CPU cores for cache locality
- **Graceful Shutdown**: Drain pending tasks before termination

```cpp
#include <vector>
#include <queue>
#include <thread>
#include <mutex>
#include <condition_variable>
#include <functional>
#include <future>
#include <atomic>

class ThreadPool {
private:
    std::vector<std::thread> workers;
    std::queue<std::function<void()>> tasks;
    
    std::mutex queue_mutex;
    std::condition_variable cv;
    std::atomic<bool> stop{false};
    
    void worker_thread() {
        while (true) {
            std::function<void()> task;
            
            {
                std::unique_lock<std::mutex> lock(queue_mutex);
                
                // Wait for task or shutdown signal
                cv.wait(lock, [this]() {
                    return stop.load() || !tasks.empty();
                });
                
                // Exit if shutting down and no tasks remain
                if (stop.load() && tasks.empty()) {
                    return;
                }
                
                // Fetch next task
                task = std::move(tasks.front());
                tasks.pop();
            }
            
            // Execute task without holding lock
            task();
        }
    }
    
public:
    explicit ThreadPool(size_t num_threads = std::thread::hardware_concurrency()) {
        workers.reserve(num_threads);
        
        for (size_t i = 0; i < num_threads; ++i) {
            workers.emplace_back(&ThreadPool::worker_thread, this);
        }
    }
    
    ~ThreadPool() {
        shutdown();
    }
    
    // Submit task and get future for result
    template<typename F, typename... Args>
    auto submit(F&& f, Args&&... args) 
        -> std::future<typename std::invoke_result_t<F, Args...>> {
        
        using return_type = typename std::invoke_result_t<F, Args...>;
        
        // Package task with promise
        auto task = std::make_shared<std::packaged_task<return_type()>>(
            std::bind(std::forward<F>(f), std::forward<Args>(args)...)
        );
        
        std::future<return_type> result = task->get_future();
        
        {
            std::unique_lock<std::mutex> lock(queue_mutex);
            
            if (stop.load()) {
                throw std::runtime_error("ThreadPool is stopped");
            }
            
            tasks.emplace([task]() { (*task)(); });
        }
        
        cv.notify_one();
        return result;
    }
    
    // Submit void task without future
    void enqueue(std::function<void()> task) {
        {
            std::unique_lock<std::mutex> lock(queue_mutex);
            
            if (stop.load()) {
                throw std::runtime_error("ThreadPool is stopped");
            }
            
            tasks.emplace(std::move(task));
        }
        
        cv.notify_one();
    }
    
    void shutdown() {
        stop.store(true);
        cv.notify_all();
        
        for (auto& worker : workers) {
            if (worker.joinable()) {
                worker.join();
            }
        }
    }
    
    size_t pending_tasks() const {
        std::unique_lock<std::mutex> lock(queue_mutex);
        return tasks.size();
    }
};
```


## 8. Circuit Breaker Pattern

### Theory

Circuit breaker prevents cascading failures in distributed systems by failing fast when a service is unhealthy. Three states: CLOSED (normal), OPEN (failing fast), HALF_OPEN (testing recovery).[^3][^4][^5]

**State Transitions**:

- **CLOSED → OPEN**: Failure threshold exceeded
- **OPEN → HALF_OPEN**: Timeout elapsed, test if service recovered
- **HALF_OPEN → CLOSED**: Test requests succeed
- **HALF_OPEN → OPEN**: Test requests fail

```cpp
#include <mutex>
#include <chrono>
#include <functional>
#include <atomic>
#include <stdexcept>

enum class CircuitState { CLOSED, OPEN, HALF_OPEN };

class CircuitBreaker {
private:
    using Clock = std::chrono::steady_clock;
    using TimePoint = Clock::time_point;
    
    std::atomic<CircuitState> state{CircuitState::CLOSED};
    std::atomic<size_t> failure_count{0};
    std::atomic<size_t> success_count{0};
    
    const size_t failure_threshold;     // Failures before opening
    const size_t success_threshold;     // Successes to close from half-open
    const std::chrono::milliseconds timeout;  // Time before retry
    
    TimePoint last_failure_time;
    mutable std::mutex state_mutex;
    
    void recordSuccess() {
        std::lock_guard<std::mutex> lock(state_mutex);
        
        failure_count.store(0);
        
        if (state.load() == CircuitState::HALF_OPEN) {
            success_count.fetch_add(1);
            
            // Transition to CLOSED if enough successes
            if (success_count.load() >= success_threshold) {
                state.store(CircuitState::CLOSED);
                success_count.store(0);
            }
        }
    }
    
    void recordFailure() {
        std::lock_guard<std::mutex> lock(state_mutex);
        
        last_failure_time = Clock::now();
        failure_count.fetch_add(1);
        
        CircuitState current = state.load();
        
        if (current == CircuitState::HALF_OPEN) {
            // Immediate transition to OPEN on failure during testing
            state.store(CircuitState::OPEN);
            success_count.store(0);
        } else if (current == CircuitState::CLOSED) {
            // Open circuit if threshold exceeded
            if (failure_count.load() >= failure_threshold) {
                state.store(CircuitState::OPEN);
            }
        }
    }
    
    bool shouldAttemptReset() {
        std::lock_guard<std::mutex> lock(state_mutex);
        
        if (state.load() != CircuitState::OPEN) {
            return false;
        }
        
        // Check if timeout elapsed
        auto elapsed = Clock::now() - last_failure_time;
        if (elapsed >= timeout) {
            state.store(CircuitState::HALF_OPEN);
            success_count.store(0);
            failure_count.store(0);
            return true;
        }
        
        return false;
    }
    
public:
    CircuitBreaker(size_t fail_threshold, 
                   size_t success_threshold,
                   std::chrono::milliseconds timeout_ms)
        : failure_threshold(fail_threshold)
        , success_threshold(success_threshold)
        , timeout(timeout_ms)
        , last_failure_time(Clock::now()) {}
    
    // Execute function with circuit breaker protection
    template<typename Func, typename... Args>
    auto call(Func&& func, Args&&... args) 
        -> typename std::invoke_result_t<Func, Args...> {
        
        CircuitState current = state.load();
        
        // Fast fail if circuit open
        if (current == CircuitState::OPEN) {
            if (!shouldAttemptReset()) {
                throw std::runtime_error("Circuit breaker is OPEN");
            }
        }
        
        try {
            // Attempt operation
            auto result = std::invoke(std::forward<Func>(func), 
                                     std::forward<Args>(args)...);
            recordSuccess();
            return result;
        } catch (...) {
            recordFailure();
            throw;
        }
    }
    
    CircuitState getState() const {
        return state.load();
    }
    
    void reset() {
        std::lock_guard<std::mutex> lock(state_mutex);
        state.store(CircuitState::CLOSED);
        failure_count.store(0);
        success_count.store(0);
    }
};
```


## 9. Lock-Free Queue (SPSC)

### Theory

Lock-free data structures use atomic operations (CAS) to avoid mutex overhead. Single-Producer-Single-Consumer (SPSC) queues are simpler than MPMC and achieve higher throughput using atomic head/tail indices.[^6][^7][^8]

**ABA Problem**: Use tagged pointers or generation counters to detect spurious CAS successes.

```cpp
#include <atomic>
#include <vector>
#include <optional>
#include <memory>

// Single Producer Single Consumer Lock-Free Queue
template<typename T>
class SPSCQueue {
private:
    struct Node {
        T data;
        std::atomic<Node*> next{nullptr};
        
        explicit Node(T val) : data(std::move(val)) {}
    };
    
    // Padded to prevent false sharing
    struct alignas(64) PaddedAtomic {
        std::atomic<Node*> ptr{nullptr};
    };
    
    PaddedAtomic head;  // Consumer reads from head
    PaddedAtomic tail;  // Producer writes to tail
    
public:
    SPSCQueue() {
        // Dummy node to simplify logic
        Node* dummy = new Node(T{});
        head.ptr.store(dummy, std::memory_order_relaxed);
        tail.ptr.store(dummy, std::memory_order_relaxed);
    }
    
    ~SPSCQueue() {
        while (auto item = dequeue()) {}
        delete head.ptr.load();
    }
    
    // Producer calls this (single thread)
    void enqueue(T value) {
        Node* new_node = new Node(std::move(value));
        
        // Get current tail
        Node* prev_tail = tail.ptr.load(std::memory_order_relaxed);
        
        // Link new node
        prev_tail->next.store(new_node, std::memory_order_release);
        
        // Update tail
        tail.ptr.store(new_node, std::memory_order_relaxed);
    }
    
    // Consumer calls this (single thread)
    std::optional<T> dequeue() {
        Node* current_head = head.ptr.load(std::memory_order_relaxed);
        Node* next = current_head->next.load(std::memory_order_acquire);
        
        if (next == nullptr) {
            return std::nullopt;  // Queue empty
        }
        
        // Extract data before updating head
        T value = std::move(next->data);
        
        // Move head forward
        head.ptr.store(next, std::memory_order_relaxed);
        
        // Delete old dummy node
        delete current_head;
        
        return value;
    }
    
    bool empty() const {
        Node* current_head = head.ptr.load(std::memory_order_relaxed);
        Node* next = current_head->next.load(std::memory_order_acquire);
        return next == nullptr;
    }
};

// Bounded Lock-Free Ring Buffer (faster for fixed capacity)
template<typename T, size_t Capacity>
class SPSCRingBuffer {
private:
    static_assert((Capacity & (Capacity - 1)) == 0, 
                  "Capacity must be power of 2");
    
    std::array<T, Capacity> buffer;
    
    // Separate cache lines to prevent false sharing
    alignas(64) std::atomic<size_t> write_idx{0};
    alignas(64) std::atomic<size_t> read_idx{0};
    
    size_t mask() const { return Capacity - 1; }
    
public:
    // Producer only
    bool enqueue(const T& value) {
        size_t current_write = write_idx.load(std::memory_order_relaxed);
        size_t next_write = (current_write + 1) & mask();
        
        // Check if full (would catch up to reader)
        if (next_write == read_idx.load(std::memory_order_acquire)) {
            return false;
        }
        
        buffer[current_write] = value;
        
        // Publish write with release semantics
        write_idx.store(next_write, std::memory_order_release);
        return true;
    }
    
    // Consumer only
    std::optional<T> dequeue() {
        size_t current_read = read_idx.load(std::memory_order_relaxed);
        
        // Check if empty
        if (current_read == write_idx.load(std::memory_order_acquire)) {
            return std::nullopt;
        }
        
        T value = buffer[current_read];
        
        // Publish read with release semantics
        read_idx.store((current_read + 1) & mask(), 
                      std::memory_order_release);
        
        return value;
    }
    
    bool empty() const {
        return read_idx.load(std::memory_order_acquire) == 
               write_idx.load(std::memory_order_acquire);
    }
    
    size_t size() const {
        size_t write = write_idx.load(std::memory_order_acquire);
        size_t read = read_idx.load(std::memory_order_acquire);
        return (write - read) & mask();
    }
};
```


## 10. Semaphore Implementation

### Theory

Semaphores control access to a resource pool. Binary semaphore (0/1) is like a mutex; counting semaphore (0 to N) allows N concurrent accessors.[^9][^10][^11]

```cpp
#include <mutex>
#include <condition_variable>
#include <chrono>

// Pre-C++20 semaphore implementation
class CountingSemaphore {
private:
    size_t count;
    const size_t max_count;
    
    std::mutex mtx;
    std::condition_variable cv;
    
public:
    explicit CountingSemaphore(size_t initial_count, 
                               size_t max = SIZE_MAX)
        : count(initial_count), max_count(max) {}
    
    // Decrement count, block if zero
    void acquire() {
        std::unique_lock<std::mutex> lock(mtx);
        
        cv.wait(lock, [this]() { 
            return count > 0; 
        });
        
        --count;
    }
    
    // Try acquire without blocking
    bool try_acquire() {
        std::unique_lock<std::mutex> lock(mtx);
        
        if (count == 0) {
            return false;
        }
        
        --count;
        return true;
    }
    
    // Try acquire with timeout
    template<typename Rep, typename Period>
    bool try_acquire_for(const std::chrono::duration<Rep, Period>& timeout) {
        std::unique_lock<std::mutex> lock(mtx);
        
        if (!cv.wait_for(lock, timeout, [this]() { 
            return count > 0; 
        })) {
            return false;
        }
        
        --count;
        return true;
    }
    
    // Increment count, wake one waiter
    void release(size_t update = 1) {
        std::unique_lock<std::mutex> lock(mtx);
        
        count = std::min(count + update, max_count);
        
        // Wake waiters (one per released permit)
        for (size_t i = 0; i < update; ++i) {
            cv.notify_one();
        }
    }
    
    size_t available() const {
        std::unique_lock<std::mutex> lock(mtx);
        return count;
    }
};

// Binary semaphore (simpler, optimized for 0/1)
class BinarySemaphore {
private:
    bool signaled;
    
    std::mutex mtx;
    std::condition_variable cv;
    
public:
    explicit BinarySemaphore(bool initial = false) 
        : signaled(initial) {}
    
    void acquire() {
        std::unique_lock<std::mutex> lock(mtx);
        
        cv.wait(lock, [this]() { 
            return signaled; 
        });
        
        signaled = false;
    }
    
    bool try_acquire() {
        std::unique_lock<std::mutex> lock(mtx);
        
        if (!signaled) {
            return false;
        }
        
        signaled = false;
        return true;
    }
    
    void release() {
        std::unique_lock<std::mutex> lock(mtx);
        
        signaled = true;
        cv.notify_one();
    }
};
```


## 11. Barrier Synchronization

### Theory

Barrier synchronizes N threads at a synchronization point. All threads block until all arrive, then all proceed together. Useful for iterative algorithms with phases.[^12][^13][^14]

```cpp
#include <mutex>
#include <condition_variable>
#include <functional>

// Pre-C++20 barrier implementation
class Barrier {
private:
    const size_t thread_count;
    size_t waiting_count{0};
    size_t generation{0};  // Detect multiple barrier cycles
    
    std::mutex mtx;
    std::condition_variable cv;
    std::function<void()> completion_callback;
    
public:
    explicit Barrier(size_t count, 
                     std::function<void()> callback = nullptr)
        : thread_count(count)
        , completion_callback(std::move(callback)) {}
    
    // All threads must call this
    void arrive_and_wait() {
        std::unique_lock<std::mutex> lock(mtx);
        
        size_t current_gen = generation;
        ++waiting_count;
        
        if (waiting_count == thread_count) {
            // Last thread to arrive
            ++generation;
            waiting_count = 0;
            
            // Execute completion callback if provided
            if (completion_callback) {
                completion_callback();
            }
            
            // Wake all waiting threads
            cv.notify_all();
        } else {
            // Wait for all threads to arrive
            // Check generation to handle spurious wakeups
            cv.wait(lock, [this, current_gen]() {
                return current_gen != generation;
            });
        }
    }
    
    // Arrive and immediately drop out (don't wait)
    void arrive_and_drop() {
        std::unique_lock<std::mutex> lock(mtx);
        
        if (--waiting_count == 0) {
            ++generation;
            
            if (completion_callback) {
                completion_callback();
            }
            
            cv.notify_all();
        }
    }
};

// Reusable barrier for multiple synchronization phases
class CyclicBarrier {
private:
    const size_t thread_count;
    size_t waiting{0};
    size_t phase{0};
    
    std::mutex mtx;
    std::condition_variable cv;
    
public:
    explicit CyclicBarrier(size_t count) 
        : thread_count(count) {}
    
    void wait() {
        std::unique_lock<std::mutex> lock(mtx);
        
        size_t current_phase = phase;
        
        if (++waiting == thread_count) {
            // Last thread resets for next phase
            waiting = 0;
            ++phase;
            cv.notify_all();
        } else {
            cv.wait(lock, [this, current_phase]() {
                return phase != current_phase;
            });
        }
    }
    
    void reset() {
        std::unique_lock<std::mutex> lock(mtx);
        waiting = 0;
        ++phase;
        cv.notify_all();
    }
};
```


## 12. Thread-Safe Singleton

### Theory

Singleton pattern ensures single instance. C++11 guarantees thread-safe static initialization. Double-checked locking (DCLP) requires careful use of atomics.[^15][^16][^17]

```cpp
#include <mutex>
#include <atomic>
#include <memory>

// Meyer's Singleton (C++11 guaranteed thread-safe)
class MeyersSingleton {
public:
    static MeyersSingleton& getInstance() {
        // Thread-safe initialization guaranteed by C++11
        static MeyersSingleton instance;
        return instance;
    }
    
    MeyersSingleton(const MeyersSingleton&) = delete;
    MeyersSingleton& operator=(const MeyersSingleton&) = delete;
    
private:
    MeyersSingleton() {}
};

// Double-Checked Locking with std::atomic (C++11 safe)
class DCLPSingleton {
public:
    static DCLPSingleton* getInstance() {
        // First check without lock (fast path)
        DCLPSingleton* tmp = instance.load(std::memory_order_acquire);
        
        if (tmp == nullptr) {
            std::lock_guard<std::mutex> lock(init_mutex);
            
            // Second check with lock
            tmp = instance.load(std::memory_order_relaxed);
            if (tmp == nullptr) {
                tmp = new DCLPSingleton();
                
                // Release ensures visibility of constructed object
                instance.store(tmp, std::memory_order_release);
            }
        }
        
        return tmp;
    }
    
private:
    static std::atomic<DCLPSingleton*> instance;
    static std::mutex init_mutex;
    
    DCLPSingleton() {}
};

std::atomic<DCLPSingleton*> DCLPSingleton::instance{nullptr};
std::mutex DCLPSingleton::init_mutex;

// Call-once initialization (cleaner alternative)
class CallOnceSingleton {
public:
    static CallOnceSingleton& getInstance() {
        std::call_once(init_flag, []() {
            instance.reset(new CallOnceSingleton());
        });
        return *instance;
    }
    
private:
    static std::unique_ptr<CallOnceSingleton> instance;
    static std::once_flag init_flag;
    
    CallOnceSingleton() {}
};

std::unique_ptr<CallOnceSingleton> CallOnceSingleton::instance;
std::once_flag CallOnceSingleton::init_flag;
```
